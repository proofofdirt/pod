Build Order & MVP Definition

This guide is ordered the way we build. Strategy in one line: build the baseline MVP in-house (1) → use it to secure financiers, DFIs, and grants (2) → spend grant money on bounties and AI competitions to improve the models (3). Bounties come last because they depend on the treasury being funded. Supervisors A/B coordinate weekly and divide the work; Sprint 0 (DEV-00) locks shared interfaces.

MVP definition (the fundraising artifact): one real pilot farm verified end-to-end — plot GeoJSON captured → SAR plowing confirmed → record anchored on devnet → tranche approval event emitted → visible live on the financier dashboard, plus one livestock intake (RFID + geolocated photo) flowing the same path. When that demo runs, Phase 1 is done and DevRel takes it to financiers and grantmakers.

T1 — Sentinel-1 SAR Plowing Detection

Plowing creates a sudden decrease in SAR backscatter (soil goes from rough to smooth). We detect pixels where VV backscatter drops below –15 dB, which indicates bare prepared soil.

# agents/satellite/sar_plowing.py
import ee
from config import AOI, SCALE

def detect_plowing(year: int, month: int) -> dict:
    """
    Returns plowed area in hectares for UKE Welega AOI.
    Best run for Nov-Dec (harvest window) and Jan-Mar (land prep).
    """
    start = ee.Date.fromYMD(year, month, 1)
    end   = start.advance(1, 'month')

    s1 = (ee.ImageCollection('COPERNICUS/S1_GRD')
          .filter(ee.Filter.eq('instrumentMode', 'IW'))
          .filter(ee.Filter.listContains('transmitterReceiverPolarisation', 'VV'))
          .filter(ee.Filter.eq('orbitProperties_pass', 'DESCENDING'))
          .filterBounds(AOI)
          .filterDate(start, end)
          .select('VV')
          .mean())  # mean composite for the month

    # Pixels below -15 dB = bare/plowed soil
    plowing_mask = s1.lt(-15)

    # Calculate area
    area_m2 = (plowing_mask
               .multiply(ee.Image.pixelArea())
               .reduceRegion(
                   reducer=ee.Reducer.sum(),
                   geometry=AOI,
                   scale=SCALE,
                   maxPixels=1e10
               ).get('VV'))

    plowed_ha = float(area_m2.getInfo()) / 10000

    return {
        "period": f"{year}-{month:02d}",
        "plowed_ha": round(plowed_ha, 2),
        "oracle_value": int(plowed_ha * 100),  # scaled for Switchboard (2 decimals)
        "source": "Sentinel-1 GRD VV <-15dB",
        "quality": "confirmed"
    }

T2 — NDVI Crop Health & T3 Yield Prediction

# agents/satellite/ndvi_yield.py
import ee, numpy as np
from sklearn.linear_model import LinearRegression
from config import AOI, SCALE

def compute_ndvi(year: int, month: int) -> dict:
    """Compute mean NDVI for AOI. Returns 0-1 float and scaled integer."""
    start = ee.Date.fromYMD(year, month, 1)
    end   = start.advance(1, 'month')

    s2 = (ee.ImageCollection('COPERNICUS/S2_SR_HARMONIZED')
          .filterBounds(AOI)
          .filterDate(start, end)
          .filter(ee.Filter.lt('CLOUDY_PIXEL_PERCENTAGE', 20))
          .median())

    ndvi = s2.normalizedDifference(['B8', 'B4']).rename('NDVI')

    mean_ndvi = (ndvi.reduceRegion(
        reducer=ee.Reducer.mean(),
        geometry=AOI, scale=SCALE, maxPixels=1e10
    ).get('NDVI'))

    ndvi_val = float(mean_ndvi.getInfo() or 0)

    return {
        "period": f"{year}-{month:02d}",
        "mean_ndvi": round(ndvi_val, 4),
        "health_score": int(ndvi_val * 100),   # 0-100 scale
        "oracle_value": int(ndvi_val * 10000), # 4 decimal precision for Switchboard
        "interpretation": (
            "poor" if ndvi_val < 0.2 else
            "moderate" if ndvi_val < 0.4 else
            "good" if ndvi_val < 0.6 else "excellent"
        )
    }

# T3 — Yield prediction (simple linear model on NDVI peak)
# Calibrate with historical Ethiopia corn yield data from FAO
# FAO: Ethiopia average corn yield ~3,200 kg/ha, range 1,800–5,200
YIELD_MODEL_SLOPE     = 8500   # kg/ha per NDVI unit (empirical)
YIELD_MODEL_INTERCEPT = -500   # kg/ha base

def predict_yield(peak_ndvi: float) -> int:
    """Returns predicted yield in kg/ha based on peak NDVI reading."""
    predicted = YIELD_MODEL_SLOPE * peak_ndvi + YIELD_MODEL_INTERCEPT
    return max(0, int(predicted))  # kg/ha, cannot be negative

T4 — Publish Results to Switchboard

# agents/satellite/run_pipeline.py
import subprocess, os
from datetime import datetime
from sar_plowing import detect_plowing
from ndvi_yield import compute_ndvi, predict_yield

def run_and_publish(year: int, month: int):
    # 1. Run satellite analysis
    sar_result  = detect_plowing(year, month)
    ndvi_result = compute_ndvi(year, month)
    yield_val   = predict_yield(ndvi_result["mean_ndvi"])

    print(f"SAR: {sar_result['plowed_ha']} ha plowed")
    print(f"NDVI: {ndvi_result['mean_ndvi']} ({ndvi_result['interpretation']})")
    print(f"Yield prediction: {yield_val} kg/ha")

    # 2. Publish to Switchboard
    feeds = [
        (os.getenv("FEED_SAR_PROOF"),   sar_result["oracle_value"]),
        (os.getenv("FEED_NDVI"),         ndvi_result["oracle_value"]),
        (os.getenv("FEED_YIELD_PRED"),   yield_val),
    ]
    for pubkey, value in feeds:
        if not pubkey: continue
        subprocess.run([
            "sb", "aggregator", "update", pubkey,
            "--keypair", os.getenv("KEYPAIR_PATH"),
            "--cluster", os.getenv("SOLANA_CLUSTER", "devnet"),
            "--value", str(value),
        ], check=True)

if __name__ == "__main__":
    now = datetime.utcnow()
    run_and_publish(now.year, now.month)

# Load latest data file (generated by price agent)
with open("data/latest.json") as f:
    data = json.load(f)

FEEDS = {
    "POD_COMMODITY_SHEEP_KSA_LW": os.getenv("FEED_SHEEP_KSA"),
    "POD_COMMODITY_GOAT_KSA_LW":  os.getenv("FEED_GOAT_KSA"),
    "POD_COMMODITY_CORN_KSA":      os.getenv("FEED_CORN_KSA"),
    "POD_COMMODITY_SORGHUM_KSA":   os.getenv("FEED_SORGHUM_KSA"),
    "POD_SAR_PLOWING_PROOF":       os.getenv("FEED_SAR_PROOF"),
    "POD_NDVI_HEALTH_SCORE":       os.getenv("FEED_NDVI"),
}

def get_scaled_value(feed_id: str) -> int:
    for feed in data["feeds"]:
        if feed["feed_id"] == feed_id:
            v = feed["value"]
            # prices < 100: USD/kg → ×10000; prices >= 100: USD/ton → ×100
            return int(v * 10000) if v < 100 else int(v * 100)
    raise ValueError(f"Feed {feed_id} not found in data")

for feed_id, pubkey in FEEDS.items():
    if not pubkey:
        print(f"SKIP {feed_id} — pubkey not set in .env")
        continue
    value = get_scaled_value(feed_id)
    result = subprocess.run([
        "sb", "aggregator", "update", pubkey,
        "--keypair", KEYPAIR,
        "--cluster", NETWORK,
        "--value", str(value),
    ], capture_output=True, text=True)
    status = "✓" if result.returncode == 0 else "✗"
    print(f"{status} {feed_id}: {value} (raw) | returncode={result.returncode}")
    if result.returncode != 0:
        print("  STDERR:", result.stderr[:200])

        submission_type TEXT NOT NULL,   -- 'field_agent' | 'sar_ground_truth'
        latitude        REAL,
        longitude       REAL,
        accuracy_m      REAL,
        photo_uri       TEXT,            -- local file path
        form_data       TEXT,            -- JSON blob of form fields
        status          TEXT DEFAULT 'pending',  -- pending|uploading|synced|failed
        retry_count     INTEGER DEFAULT 0,
        created_at      INTEGER,         -- unix timestamp
        synced_at       INTEGER          -- null until synced
      )
    `);

    // Cached task list (so agent can browse without network)
    // NOTE: agents are paid in ETB by Turmi — reward shown in Birr, no tokens/wallets
    tx.executeSql(`
      CREATE TABLE IF NOT EXISTS tasks_cache (
        task_id         TEXT PRIMARY KEY,
        title           TEXT,
        description     TEXT,
        reward_etb      INTEGER,         -- payout in Ethiopian Birr via Turmi payroll
        task_type       TEXT,
        expires_at      INTEGER,
        cached_at       INTEGER
      )
    `);

    // Agent's own assigned tasks
    tx.executeSql(`
      CREATE TABLE IF NOT EXISTS my_tasks (
        task_id TEXT PRIMARY KEY,
        status TEXT,
        claimed_at INTEGER,
        submission_queue_id INTEGER
      )
    `);
  });
}

// Write a new submission to the local queue
export function queueSubmission(submission: SubmissionData): Promise<number> {
  return new Promise((resolve, reject) => {
    db.transaction(tx => {
      tx.executeSql(
        `INSERT INTO submission_queue
         (task_id, submission_type, latitude, longitude, accuracy_m,
          photo_uri, form_data, created_at)
         VALUES (?, ?, ?, ?, ?, ?, ?, ?)`,
        [
          submission.taskId, submission.type,
          submission.gps?.latitude, submission.gps?.longitude, submission.gps?.accuracy,
          submission.photoUri, JSON.stringify(submission.formData),
          Date.now(),
        ],
        (_, result) => resolve(result.insertId),
        (_, err) => { reject(err); return false; }
      );
    });
  });
}

import * as Location from 'expo-location';
import * as ImagePicker from 'expo-image-picker';

export async function captureGPS(): Promise<GPSResult> {
  const { status } = await Location.requestForegroundPermissionsAsync();
  if (status !== 'granted') throw new Error('Location permission denied');

  const loc = await Location.getCurrentPositionAsync({
    accuracy: Location.Accuracy.High,  // best accuracy, may take a few seconds
  });

  return {
    latitude:  loc.coords.latitude,
    longitude: loc.coords.longitude,
    accuracy_m: loc.coords.accuracy,
    timestamp: loc.timestamp,
  };
}

export async function capturePhoto(): Promise<string | null> {
  const { status } = await ImagePicker.requestCameraPermissionsAsync();
  if (status !== 'granted') return null;

  const result = await ImagePicker.launchCameraAsync({
    mediaTypes: ImagePicker.MediaTypeOptions.Images,
    quality: 0.6,       // compress — saves space + faster upload
    exif: true,         // keep EXIF for timestamp verification
  });

  if (result.canceled) return null;
  return result.assets[0].uri;  // local file URI
}

async function syncQueue() {
  const pending = await getPendingSubmissions();  // from SQLite
  if (pending.length === 0) return;

  for (const submission of pending) {
    try {
      await markUploading(submission.id);

      // Upload photo to IPFS if present
      let ipfsUri = null;
      if (submission.photo_uri) {
        ipfsUri = await uploadPhoto(submission.photo_uri);
      }

      // Submit to backend API
      await fetch(`${API_BASE}/tasks/${submission.task_id}/evidence`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          submissionType: submission.submission_type,
          gps:    { lat: submission.latitude, lng: submission.longitude },
          ipfsUri,
          formData: JSON.parse(submission.form_data),
          agentId: await getStoredAgentId(),  // phone-number-based ID — agents have NO wallets
        }),
      });

      await markSynced(submission.id);
    } catch (err) {
      await markFailed(submission.id);  // will retry on next sync
      console.error('Sync failed for:', submission.id, err);
    }
  }
}

Vault Records Program

Anchor vault program — on-chain financing records and tranche approval logic tied to oracle price and satellite proofs. The chain verifies and audits; fiat moves off-chain via the KSA SPV. Greenfield — build from scratch.

Track A Rust · Anchor · SPL
V1 architecture — read before coding: Ethiopian law prohibits stablecoin/cryptocurrency transactions. The vault holds no funds on-chain. Financiers — buyers (offtake prepayment), Ethiopian FIs (oracle-collateralized loans), DFIs (impact debt) — commit fiat via the SPV or partner FI; the SPV/FI authority records each confirmed financing on-chain as a plain record (no tokens are minted — the project does not issue a token). When oracle conditions are met (KSA sheep price ≥ threshold AND SAR confirms plowing AND NDVI confirms crop health), the program approves the tranche — it emits an on-chain approval event with a disbursement hash, and SPV ops execute a fiat bank wire to Turmi (or another Ethiopian operator) via the Ethiopian financial institution partner. In V1 the chain carries verified data only: oracle records, subscriptions, approvals, audit logs. Fully on-chain settlement is a V2 goal, contingent on Ethiopian regulation.
No token: the project does not mint an SPL token. The vault program works entirely with plain account records. (A non-transferable receipt instrument may be revisited later only if fiat recordkeeping requires it.)

Step 3 — Tranche Approval Logic (no token transfer)

// programs/pod-vault/src/instructions/approve.rs
pub fn handler(ctx: Context<ApproveTranche>, disbursement_hash: [u8; 32]) -> Result<()> {
    let vault = &mut ctx.accounts.vault;
    let tranche_idx = vault.current_tranche as usize;

    require!(tranche_idx < 5, VaultError::AllTranchesApproved);
    let tranche = vault.tranches[tranche_idx];
    require!(!tranche.approved, VaultError::TrancheAlreadyApproved);

    // --- Read Switchboard oracle feeds ---
    let sheep_feed = &ctx.accounts.sheep_price_feed.load()?;
    let sheep_price = sheep_feed.get_result()?.try_into_i128()? as u64;
    // sheep_price is USD/kg * 10000 (4 decimal places in oracle)

    let sar_feed = &ctx.accounts.sar_proof_feed.load()?;
    let plowed_ha = sar_feed.get_result()?.try_into_i128()? as u64;

    let ndvi_feed = &ctx.accounts.ndvi_feed.load()?;
    let ndvi_score = ndvi_feed.get_result()?.try_into_i128()? as u8;

    // --- Verify conditions ---
    require!(
        sheep_price >= tranche.min_sheep_price_bps,
        VaultError::SheeepPriceTooLow
    );
    require!(
        plowed_ha >= tranche.min_sar_hectares,
        VaultError::InsufficientPlowedArea
    );
    if tranche.min_ndvi_score > 0 {
        require!(ndvi_score >= tranche.min_ndvi_score, VaultError::NDVITooLow);
    }

    // --- NO on-chain money movement (Ethiopia: fiat only) ---
    // Record approval + disbursement hash; SPV ops dashboard watches this
    // event and executes the fiat bank wire to Turmi off-chain.
    vault.tranches[tranche_idx].approved = true;
    vault.tranches[tranche_idx].approved_at = Clock::get()?.unix_timestamp;
    vault.tranches[tranche_idx].disbursement_hash = disbursement_hash;
    vault.current_tranche += 1;

    emit!(TrancheApprovedEvent {
        tranche: tranche_idx as u8,
        usd_cents: tranche.usd_cents,
        disbursement_hash,           // audit link to the off-chain fiat wire record
        sheep_price,
        plowed_ha,
        ndvi_score,
    });

    Ok(())
}
Why no transfer: the on-chain program is the verification and audit layer (V1). The fiat wire itself is executed by PODRWA SPV (KSA) through the Ethiopian financial institution partner; the disbursement_hash ties the wire record immutably to the on-chain approval. Direct on-chain settlement moves to V2 when Ethiopian regulation allows.

import { AggregatorAccount, SwitchboardProgram } from '@switchboard-xyz/solana.js';

export async function fetchOraclePrices() {
  const sbProgram = await SwitchboardProgram.load('devnet', connection);
  const feedIds = {
    sheep_ksa:  process.env.NEXT_PUBLIC_FEED_SHEEP_KSA,
    goat_ksa:   process.env.NEXT_PUBLIC_FEED_GOAT_KSA,
    sar_proof:  process.env.NEXT_PUBLIC_FEED_SAR_PROOF,
    ndvi:       process.env.NEXT_PUBLIC_FEED_NDVI,
  };

  const results: Record<string, number> = {};
  for (const [key, pubkeyStr] of Object.entries(feedIds)) {
    if (!pubkeyStr) continue;
    const agg = new AggregatorAccount(sbProgram, new PublicKey(pubkeyStr));
    const val = await agg.fetchLatestValue();
    results[key] = val?.toNumber() ?? 0;
  }
  return results;
}

// scripts/pay-bounty.ts — run by treasury multisig signers (USDC)
import { getOrCreateAssociatedTokenAccount, transfer } from '@solana/spl-token';

async function payBounty(recipient: PublicKey, amountUsd: number) {
  const ata = await getOrCreateAssociatedTokenAccount(connection, payer, USDC_MINT, recipient);
  const sig = await transfer(connection, payer, treasuryAta.address, ata.address,
                             multisigAuthority, amountUsd * 1e6);  // USDC 6 decimals
  return sig;  // store in registry: payoutTxSig — this IS the audit record
}
Why this is enough: every payout is already an immutable on-chain record (the transfer signature). The registry + GitHub issue thread supply the context. An escrow state machine added complexity without adding trust at this scale; revisit on-chain escrow in V2 if bounty volume demands it.
